CREATE WAREHOUSE VARSHINI_SALES_WAREHOUSE
WITH
WAREHOUSE_SIZE = 'XSMALL'
AUTO_SUSPEND = 60
AUTO_RESUME = TRUE
INITIALLY_SUSPENDED = TRUE;

CREATE DATABASE VARSHINI_DB;
SHOW DATABASES;

CREATE SCHEMA VARSHINI_DB.SALES_SCHEMA;
SHOW SCHEMAS IN DATABASE VARSHINI_DB;
USE WAREHOUSE VARSHINI_SALES_WAREHOUSE;
USE DATABASE VARSHINI_DB;
USE SCHEMA SALES_SCHEMA;

SELECT
    CURRENT_WAREHOUSE(),
    CURRENT_DATABASE(),
    CURRENT_SCHEMA(),
    CURRENT_ROLE();

    CREATE TABLE SALES_DATA (
    ORDERID VARCHAR(20),
    ORDERDATE DATE,
    CUSTOMER VARCHAR(50),
    PRODUCT VARCHAR(100),
    CATEGORY VARCHAR(50),
    QUANTITY NUMBER(10,0),
    UNITPRICE NUMBER(10,2),
    SALESAMOUNT NUMBER(12,2),
    REGION VARCHAR(30),
    SALESCHANNEL VARCHAR(30)
);

CREATE OR REPLACE FILE FORMAT SALES_CSV_FORMAT
TYPE = 'CSV'
SKIP_HEADER = 1
FIELD_OPTIONALLY_ENCLOSED_BY = '"'
EMPTY_FIELD_AS_NULL = TRUE
NULL_IF = ('NULL', 'null');

CREATE OR REPLACE STAGE SALES_STAGE
FILE_FORMAT = SALES_CSV_FORMAT;

COPY INTO SALES_DATA
FROM @SALES_STAGE
FILE_FORMAT = (FORMAT_NAME = SALES_CSV_FORMAT)
ON_ERROR = 'ABORT_STATEMENT';

SELECT
    SUM(SALESAMOUNT) AS TOTAL_REVENUE
FROM SALES_DATA;

SELECT
    COUNT(DISTINCT CUSTOMER) AS UNIQUE_CUSTOMERS
FROM SALES_DATA;
DROP DATABASE database_name;
